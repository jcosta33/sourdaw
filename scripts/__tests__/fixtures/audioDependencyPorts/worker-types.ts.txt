export async function probe(): Promise<[GPUAdapter, GPUDevice, GPUAdapterInfo, GPUFeatureName[]] | null> {
 const adapter = await navigator.gpu.requestAdapter({powerPreference: 'high-performance', featureLevel: 'core'});
 if (!adapter) return null;
 const features: GPUFeatureName[] = ['timestamp-query', 'bgra8unorm-storage'];
 const device = await adapter.requestDevice({requiredFeatures: features});
 const info: GPUAdapterInfo = adapter.info;
 return [adapter, device, info, features];
}
